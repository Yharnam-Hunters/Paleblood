cmake_minimum_required(VERSION 3.20)
project(paleblood_logical_cpu C)
if(APPLE)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Darwin" OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "x86_64")
        message(FATAL_ERROR "Owned runtime component tests require explicit single x86_64 Darwin architecture")
    endif()
elseif(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
    message(FATAL_ERROR "This isolated owned test proposal supports whole x86-64 Darwin or Linux only")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Owned tests require a whole 64-bit process")
endif()
if(CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Owned tests use Release" FORCE)
elseif(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Owned tests use Release" FORCE)
elseif(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Owned tests require Release")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Werror)
enable_testing()
get_filename_component(PB_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(pb_logical_cpu STATIC "${PB_ROOT}/runtime/logical_cpu.c")
target_include_directories(pb_logical_cpu PUBLIC "${PB_ROOT}/runtime/include")
add_executable(test_logical_cpu "${PB_ROOT}/test/test_logical_cpu.c")
target_link_libraries(test_logical_cpu PRIVATE pb_logical_cpu)
add_test(NAME logical_cpu_slots COMMAND test_logical_cpu)
set_tests_properties(logical_cpu_slots PROPERTIES TIMEOUT 10)
if(APPLE)
    # Includes exact component with test-only sysctl alias; do not link the real component too.
    add_executable(test_logical_cpu_query_failures "${PB_ROOT}/test/test_logical_cpu_query_failures.c")
    target_include_directories(test_logical_cpu_query_failures PRIVATE "${PB_ROOT}/runtime/include")
    add_test(NAME logical_cpu_query_failures COMMAND test_logical_cpu_query_failures)
    # One real count read as an ordinary owned component test, not a capability-probe executable.
    add_executable(test_logical_cpu_host_count "${PB_ROOT}/test/test_logical_cpu_host_count.c")
    target_link_libraries(test_logical_cpu_host_count PRIVATE pb_logical_cpu)
    add_test(NAME logical_cpu_host_count COMMAND test_logical_cpu_host_count)
    set_tests_properties(logical_cpu_query_failures logical_cpu_host_count PROPERTIES TIMEOUT 10)
endif()

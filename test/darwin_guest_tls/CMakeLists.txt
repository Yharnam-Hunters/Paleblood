cmake_minimum_required(VERSION 3.20)
project(paleblood_darwin_guest_tls C)

if(NOT APPLE OR NOT CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    message(FATAL_ERROR "This isolated owned-TSD test project requires macOS.")
endif()
if(CMAKE_OSX_ARCHITECTURES)
    set(PB_GUEST_TLS_ARCH "${CMAKE_OSX_ARCHITECTURES}")
else()
    set(PB_GUEST_TLS_ARCH "${CMAKE_SYSTEM_PROCESSOR}")
endif()
if(NOT PB_GUEST_TLS_ARCH STREQUAL "x86_64")
    message(FATAL_ERROR "Owned GS-slot tests require only x86_64; pass -DCMAKE_OSX_ARCHITECTURES=x86_64. ARM64/universal builds are unsupported.")
endif()
if(CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Owned tests use Release" FORCE)
elseif(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Owned tests use Release" FORCE)
elseif(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Owned TSD tests require CMAKE_BUILD_TYPE=Release.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Werror)
find_package(Threads REQUIRED)
enable_testing()

get_filename_component(PB_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(guest_tls_owned STATIC "${PB_SOURCE_ROOT}/runtime/guest_tls_darwin.c")
target_include_directories(guest_tls_owned PUBLIC "${PB_SOURCE_ROOT}/runtime/include")
target_link_libraries(guest_tls_owned PUBLIC Threads::Threads)

add_executable(test_guest_tls_darwin "${PB_SOURCE_ROOT}/test/test_guest_tls_darwin.c")
target_link_libraries(test_guest_tls_darwin PRIVATE guest_tls_owned)
# This test includes the implementation with fake public APIs; do not also link the real unit.
add_executable(test_guest_tls_failures "${PB_SOURCE_ROOT}/test/test_guest_tls_failures.c")
target_include_directories(test_guest_tls_failures PRIVATE "${PB_SOURCE_ROOT}/runtime/include")
target_link_libraries(test_guest_tls_failures PRIVATE Threads::Threads)
add_test(NAME darwin_guest_tls_owned COMMAND test_guest_tls_darwin)
add_test(NAME darwin_guest_tls_failures COMMAND test_guest_tls_failures)
set_tests_properties(darwin_guest_tls_owned darwin_guest_tls_failures PROPERTIES TIMEOUT 10)
